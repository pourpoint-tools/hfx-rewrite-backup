from pathlib import Path

import pyarrow as pa
import pyarrow.parquet as pq
import pytest
from geoparquet_bulk import check_census, scan


def test_native_streamed_census_and_expected_checks(tmp_path: Path) -> None:
    schema = pa.schema(
        [
            ("id", pa.int64()),
            ("level", pa.int16()),
            ("bbox", pa.struct([(n, pa.float32()) for n in ("xmin", "ymin", "xmax", "ymax")])),
            ("source_length", pa.float64()),
            ("geometry", pa.binary()),
        ]
    )
    table = pa.Table.from_pylist(
        [
            {
                "id": i + 1,
                "level": i % 2,
                "bbox": {"xmin": 0.0, "ymin": 0.0, "xmax": 1.0, "ymax": 1.0},
                "source_length": None if i == 1 else 3.0,
                "geometry": b"not projected",
            }
            for i in range(3)
        ],
        schema=schema,
    )
    path = tmp_path / "tiny.parquet"
    pq.write_table(table, path, row_group_size=2)
    with pq.ParquetFile(path, pre_buffer=True) as reader:
        result = scan(reader)
    assert result["rows"] == 3 and result["emitted_batches"] == 1
    assert result["by_tier_count"] == {"0": 2, "1": 1}
    assert result["null_counts"]["source_length"] == 1
    assert result["id_sum_diagnostic"] == 6
    check_census(result, 3, result)
    with pytest.raises(ValueError, match="tier"):
        check_census(result, 3, {**result, "by_tier_count": {"0": 3}})
    with pytest.raises(ValueError, match="null"):
        check_census(result, 3, {**result, "null_counts": {}})
