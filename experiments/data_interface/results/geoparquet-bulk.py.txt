"""One private streamed native GeoParquet projected census; not maintained reader code."""

from __future__ import annotations

import argparse
import hashlib
import json
from pathlib import Path
from time import perf_counter
from typing import Any, cast

import numpy as np
import pyarrow as pa
import pyarrow.compute as pc
import pyarrow.parquet as pq

from experiments.data_interface.build import peak_rss
from experiments.data_interface.geoparquet import GeoParquetAccess
from experiments.data_interface.record_probe import TRANSPORT

SOURCE_SHA = "b9a38083131dcf1a4c7c4e73ebe9a97dff2a4b7ef91a8df53f8bc38cadfa48db"
DERIVATIVE_SHA = "794c254cde8747bfb75258c004d005a35574d30a1f9bbd5163369279c684bc44"
CANDIDATE_SHA = "155535aea34d16e8369ec0db416d2f58058911553c28e9031148607bb7511049"
COLUMNS = ["id", "level", "bbox", "source_length"]


def scan(reader: pq.ParquetFile) -> dict[str, Any]:
    rows = id_sum = max_buffer = logical = batches = 0
    nulls = dict.fromkeys([*COLUMNS, *["bbox." + n for n in ("xmin", "ymin", "xmax", "ymax")]], 0)
    tiers: dict[str, int] = {}
    for batch in reader.iter_batches(batch_size=65536, columns=COLUMNS, use_threads=False):
        batches += 1
        rows += batch.num_rows
        logical += batch.nbytes
        max_buffer = max(max_buffer, batch.get_total_buffer_size())
        id_sum += int(pc.sum(batch["id"]).as_py())
        for name in COLUMNS:
            nulls[name] += batch[name].null_count
        levels, counts = np.unique(batch["level"].to_numpy(), return_counts=True)
        for level, count in zip(levels.tolist(), counts.tolist(), strict=True):
            tiers[str(level)] = tiers.get(str(level), 0) + int(count)
        bounds = cast(pa.StructArray, batch["bbox"])
        coords = []
        for name in ("xmin", "ymin", "xmax", "ymax"):
            nulls["bbox." + name] += bounds.field(name).null_count
            coords.append(bounds.field(name).to_numpy())
        if (
            any(np.any(~np.isfinite(values)) for values in coords)
            or np.any(coords[0] > coords[2])
            or np.any(coords[1] > coords[3])
        ):
            raise ValueError("invalid projected bbox")
        del batch, bounds, coords, levels, counts
    return {
        "rows": rows,
        "emitted_batches": batches,
        "id_sum_diagnostic": id_sum,
        "null_counts": nulls,
        "by_tier_count": tiers,
        "projected_logical_bytes": logical,
        "max_projected_batch_buffer_bytes": max_buffer,
    }


def check_census(result: dict[str, Any], expected_rows: int, expected: dict[str, Any]) -> None:
    if result["rows"] != expected_rows:
        raise ValueError("global source row count differs")
    if result["by_tier_count"] != expected["by_tier_count"]:
        raise ValueError("source tier census differs")
    if result["null_counts"] != expected["null_counts"]:
        raise ValueError("source null census differs")


def measure(bundle: Path, candidate: Path, source: Path, witness: Path) -> dict[str, Any]:
    source_raw = (source / "manifest.json").read_bytes()
    if hashlib.sha256(source_raw).hexdigest() != SOURCE_SHA:
        raise ValueError("source manifest differs")
    expected_rows = json.loads(source_raw)["rivers"]["count"]
    expected_raw = witness.read_bytes()
    expected = json.loads(expected_raw)
    if not expected["complete_result"] or expected["source_manifest_sha256"] != SOURCE_SHA:
        raise ValueError("independent census source differs")
    started = perf_counter()
    raw = (bundle / "manifest.json").read_bytes()
    if hashlib.sha256(raw).hexdigest() != DERIVATIVE_SHA:
        raise ValueError("derivative declaration differs")
    declaration = json.loads(raw)
    if (
        declaration["verification"]["status"] != "passed"
        or declaration["binding"]["source_manifest_sha256"] != CANDIDATE_SHA
    ):
        raise ValueError("derived source binding differs")
    access = GeoParquetAccess(str(bundle / declaration["artifact"]), str(candidate), declaration["binding"])
    try:
        opening_s = perf_counter() - started
        opening = access.stats()
        candidate_bytes = (candidate / "manifest.json").stat().st_size
        plain_manifest = {"local_manifest_read_calls": 2, "local_manifest_read_bytes": len(raw) + candidate_bytes}
        report: dict[str, Any] = {
            "complete_result": False,
            "columns": COLUMNS,
            "source_manifest_sha256": SOURCE_SHA,
            "derivative_manifest_sha256": DERIVATIVE_SHA,
            "candidate_manifest_sha256": CANDIDATE_SHA,
            "independent_census_sha256": hashlib.sha256(expected_raw).hexdigest(),
            "opening_s": opening_s,
            "opening": {**opening, **plain_manifest},
            "physical_row_groups": access.reader.num_row_groups,
            "phases": [],
            "policy": "ParquetFile.iter_batches batch_size65536/use_threadsFalse/pre_bufferTrue; same reader twice; no application batch cache or full download; no geometry decode; native serialized prebuffer retention unmeasured",
            "memory_note": "max batch Arrow buffer is not total live/native serialized cache or RSS; lifetime RSS includes imports/validation and inherited high-water",
            "validation_scope": "source global/tier/null census; ID sum diagnostic not multiset proof; complete exact preservation separately verified",
            "concurrency": "unrelated upload PID61090 concurrent; elapsed nonisolated; no other root performance trials",
        }
        previous = opening
        first_result = None
        for repeat in range(2):
            start = perf_counter()
            result = scan(access.reader)
            elapsed = perf_counter() - start
            stats = access.stats()
            check_census(result, expected_rows, expected["phases"][0])
            if first_result is not None:
                for name in ("rows", "by_tier_count", "null_counts", "id_sum_diagnostic"):
                    if first_result[name] != result[name]:
                        raise ValueError("repeat census differs")
            first_result = result
            transport = {key: int(stats[key]) - int(previous[key]) for key in TRANSPORT}
            report["phases"].append(
                {
                    **result,
                    "repeat": repeat,
                    "scan_s": elapsed,
                    "total_to_result_s": elapsed + (opening_s if repeat == 0 else 0),
                    "scan_transport": transport,
                    "cumulative_transport": {**{key: int(stats[key]) for key in TRANSPORT}, **plain_manifest},
                    "complete_scan": True,
                    "process_lifetime_peak_rss_bytes": peak_rss(),
                }
            )
            previous = stats
        report.update({"complete_result": True, "final": access.stats(), "process_lifetime_peak_rss_bytes": peak_rss()})
        return report
    finally:
        access.close()


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--bundle", type=Path, required=True)
    parser.add_argument("--candidate", type=Path, required=True)
    parser.add_argument("--source", type=Path, required=True)
    parser.add_argument("--witness", type=Path, required=True)
    parser.add_argument("--report", type=Path, required=True)
    args = parser.parse_args()
    if any(args.report.resolve().is_relative_to(root.resolve()) for root in (args.bundle, args.candidate, args.source)):
        raise ValueError("report must be outside bundles")
    script_sha = hashlib.sha256(Path(__file__).read_bytes()).hexdigest()
    with args.report.open("x") as stream:
        try:
            result = measure(args.bundle, args.candidate, args.source, args.witness)
        except Exception as error:
            stream.write(
                json.dumps({"complete_result": False, "error": str(error), "script_sha256": script_sha}) + "\n"
            )
            raise
        result.update(
            {
                "script_sha256": script_sha,
                "versions": {"pyarrow": pa.__version__, "numpy": np.__version__},
                "reader_commit": "44f1346ce298ee241841c890a557750346c3f122",
            }
        )
        stream.write(json.dumps(result, indent=2) + "\n")
    print(json.dumps({"complete_result": True, "report": str(args.report)}))


if __name__ == "__main__":
    main()
