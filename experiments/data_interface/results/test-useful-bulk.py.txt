import json
from pathlib import Path

import pyarrow as pa
import pyarrow.ipc as ipc
import pytest
import shapely
from useful_bulk import census, immediate

from experiments.data_interface.features import build_features
from hfx.candidates import fingerprint


@pytest.fixture
def bundles(tmp_path: Path) -> tuple[Path, Path]:
    bbox = pa.struct([pa.field(n, pa.float32(), False) for n in ("xmin", "ymin", "xmax", "ymax")])
    schema = pa.schema(
        [
            pa.field("id", pa.int64(), False),
            pa.field("level", pa.int16(), False),
            pa.field("bbox", bbox, False),
            pa.field("source_length", pa.float64()),
            pa.field("upstream_node_id", pa.int64(), False),
            pa.field("downstream_node_id", pa.int64(), False),
            pa.field("geometry", pa.binary(), False),
        ],
        metadata={b"native": b"exact"},
    )
    rows = []
    for identity, level, length, up, down, x in [
        (90, 1, None, 3, 2, 10.0),
        (2, 0, 4.5, 2, 5, 0.0),
        (700, 1, 2.0, 4, 2, -10.0),
    ]:
        rows.append(
            {
                "id": identity,
                "level": level,
                "bbox": {"xmin": x, "ymin": 0.0, "xmax": x + 1, "ymax": 1.0},
                "source_length": length,
                "upstream_node_id": up,
                "downstream_node_id": down,
                "geometry": shapely.to_wkb(shapely.Point(x, 0)),
            }
        )
    raw = tmp_path / "raw.arrow"
    with ipc.new_file(str(raw), schema) as writer:
        writer.write_table(pa.Table.from_pylist(rows, schema=schema), max_chunksize=2)
    source = tmp_path / "source"
    source.mkdir()
    declaration = build_features(raw, source, "rivers", rows=2)
    artifacts = {p.name: fingerprint(p) for p in source.iterdir()}
    manifest = {"format_version": "0.4.0", "rivers": {**declaration, "count": 3}, "artifacts": artifacts}
    (source / "manifest.json").write_text(json.dumps(manifest))
    candidate = tmp_path / "candidate"
    candidate.mkdir()
    candidate_decl = build_features(raw, candidate, "rivers", rows=1)
    relation_schema = pa.schema(
        [
            pa.field("id", pa.int64(), False),
            pa.field("incoming_river_ids", pa.list_(pa.int64()), False),
            pa.field("outgoing_river_ids", pa.list_(pa.int64()), False),
        ]
    )
    with ipc.new_file(str(candidate / "relations.arrow"), relation_schema) as writer:
        writer.write_table(
            pa.table(
                {"id": [2], "incoming_river_ids": [[90, 700]], "outgoing_river_ids": [[2]]}, schema=relation_schema
            )
        )
    (candidate / "relations-directory.json").write_text(
        json.dumps({"batches": [{"batch": 0, "min_id": 2, "max_id": 2, "rows": 1}]})
    )
    artifacts = {p.name: fingerprint(p) for p in candidate.iterdir()}
    import hashlib

    candidate_manifest = {
        "experimental": True,
        "input_manifest_sha256": hashlib.sha256((source / "manifest.json").read_bytes()).hexdigest(),
        "families": {"rivers": {**candidate_decl, "count": 3}},
        "relationships": {"river_nodes": {"data": "relations.arrow", "directory": "relations-directory.json"}},
        "artifacts": artifacts,
    }
    (candidate / "manifest.json").write_text(json.dumps(candidate_manifest))
    return source, candidate


def test_streamed_census_reorder_null_tier_and_id_diagnostic(bundles: tuple[Path, Path]) -> None:
    source, candidate = bundles
    for bundle in [source, candidate]:
        report = census(str(bundle), "rivers", source)
        assert report["complete_result"] is True
        for phase in report["phases"]:
            assert phase["rows"] == 3
            assert phase["id_sum_diagnostic"] == 792
            assert phase["null_counts"]["source_length"] == 1
            assert phase["by_tier_count"] == {"0": 1, "1": 2}
            assert phase["max_projected_batch_buffer_bytes"] > 0


def test_scalar_and_published_incidence_fetch_exact_native_records(bundles: tuple[Path, Path]) -> None:
    source, candidate = bundles
    for bundle in [source, candidate]:
        report = immediate(str(bundle), source)
        for phase in report["phases"]:
            assert phase["incoming_ids"] == [90, 700]
            assert phase["outgoing_ids"] == [2]
            assert phase["record_rows"] == 3
            assert phase["validation"]["semantic_equal"] is True
            assert phase["decoded_geometries"] == 3
            assert phase["total_to_result_s"] == phase["total_to_records_s"] + phase["geometry_decode_s"]
        assert report["complete_result"] is True
